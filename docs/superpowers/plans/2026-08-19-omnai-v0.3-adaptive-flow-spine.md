```ts
export const DECISION_OWNERS = ['HUMAN', 'AGENT', 'EXTERNAL'] as const;
export const DECISION_STATUSES = ['OPEN', 'BLOCKED', 'RESOLVED', 'REJECTED', 'SUPERSEDED'] as const;
export const DECISION_OPTION_STATUSES = ['VIABLE', 'REJECTED'] as const;
export const DECISION_AUTHORITIES = ['HUMAN_CONFIRMED', 'AGENT_EVIDENCE', 'EXTERNAL_CONFIRMED'] as const;
export const FLOW_SCALES = ['LOCAL', 'CHANGE', 'PROGRAM'] as const;
export const FLOW_UNCERTAINTY = ['CLEAR', 'OPEN', 'BLOCKED'] as const;
export const FLOW_TOPOLOGIES = ['SINGLE_MODULE', 'CROSS_MODULE', 'CROSS_PROJECT'] as const;
export const ARCHITECTURE_APPLICABILITIES = ['NOT_APPLICABLE', 'FOCUSED', 'FULL'] as const;
export const DELIVERY_SHAPES = ['STANDARD', 'MIGRATION', 'HIGH_RISK'] as const;
export const CAPABILITY_DISPOSITIONS = ['REQUIRED', 'CONDITIONAL', 'NOT_APPLICABLE'] as const;

export const sourceRefSchema = z.strictObject({
  kind: z.enum(SOURCE_REF_KINDS),
  ref: z.string().min(1),
  contentHash: z.string().regex(/^sha256:[0-9a-f]{64}$/),
});

const decisionAffectsSchema = z.strictObject({
  capabilities: z.array(z.enum(CAPABILITIES)).default([]),
  artifacts: z.array(z.string().min(1)).default([]),
  tasks: z.array(z.string().regex(/^TASK-\d{3}$/)).default([]),
  projects: z.array(z.string().min(1)).default([]),
  contracts: z.array(z.string().min(1)).default([]),
});

export const decisionOptionSchema = z.strictObject({
  id: z.string().regex(/^OPT-\d{2}$/),
  label: z.string().min(1),
  status: z.enum(DECISION_OPTION_STATUSES),
  consequences: z.array(z.string().min(1)).default([]),
  sourceRefs: z.array(sourceRefSchema).min(1),
});

export const decisionRecordSchema = z.strictObject({
  schemaVersion: z.literal(1),
  id: z.string().regex(/^DEC-\d{4}$/),
  changeId: z.string().regex(/^CHG-\d{4}$/),
  openedRevision: z.string().regex(/^REV-\d{4}$/),
  resolvedRevision: z.string().regex(/^REV-\d{4}$/).nullable(),
  kind: z.enum(DECISION_KINDS),
  owner: z.enum(DECISION_OWNERS),
  status: z.enum(DECISION_STATUSES),
  blocking: z.boolean(),
  question: z.string().min(1),
  options: z.array(decisionOptionSchema),
  resolution: z.strictObject({ optionId: z.string().regex(/^OPT-\d{2}$/).nullable(), summary: z.string().min(1), authority: z.enum(DECISION_AUTHORITIES), sourceRefs: z.array(sourceRefSchema).min(1) }).nullable(),
  supersededBy: z.string().regex(/^DEC-\d{4}$/).nullable(),
  affects: decisionAffectsSchema,
  sourceRefs: z.array(sourceRefSchema).min(1),
  createdAt: z.string().datetime(),
  updatedAt: z.string().datetime(),
}).superRefine((record, context) => {
  const optionIds = record.options.map((option) => option.id);
  if (new Set(optionIds).size !== optionIds.length) context.addIssue({ code: 'custom', path: ['options'], message: 'option IDs must be unique' });
  if (record.status === 'RESOLVED' && (!record.resolution || !record.resolvedRevision)) context.addIssue({ code: 'custom', path: ['resolution'], message: 'RESOLVED requires resolution and resolvedRevision' });
  if ((record.status === 'OPEN' || record.status === 'BLOCKED') && (record.resolution || record.resolvedRevision)) context.addIssue({ code: 'custom', path: ['resolution'], message: 'open decisions cannot carry a resolution' });
  if (record.status === 'SUPERSEDED' && !record.supersededBy) context.addIssue({ code: 'custom', path: ['supersededBy'], message: 'SUPERSEDED requires supersededBy' });
  if (record.resolution?.optionId && !optionIds.includes(record.resolution.optionId)) context.addIssue({ code: 'custom', path: ['resolution', 'optionId'], message: 'selected option must exist' });
});
```
